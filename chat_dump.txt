import { groq } from '@ai-sdk/groq';
import { streamText, tool } from 'ai';
import { z } from 'zod';

async function searchProducts(query: string) {
  const demoProducts = [
    { name: 'Obagi Vitamin C Serum', price: 1200, url: '/product/obagi-serum' },
    { name: 'Haku Cream', price: 550, url: '/product/haku-cream' },
    { name: 'Transino Whitening', price: 850, url: '/product/transino' }
  ];
  return demoProducts.filter(p => p.name.toLowerCase().includes(query.toLowerCase()));
}

export async function POST(req: Request) {
  try {
    const { messages } = await req.json();

    const result = streamText({
      model: groq('llama-3.3-70b-versatile'),
      messages,
      system: `You are T-bot, a polite and helpful customer support assistant for Trillion Mart. Always reply in the language the user asks (Bangla or English).`,
      tools: {
        searchProducts: tool({
          description: 'Search for products in the Trillion Mart store based on user query.',
          parameters: z.object({
            query: z.string().describe('The name or keyword of the product'),
          }),
          execute: async ({ query }) => {
            const products = await searchProducts(query);
            return { products };
          },
        }),
      },
    });

    return result.toDataStreamResponse();
  } catch (error: any) {
    console.error("API Error:", error);
    return new Response(JSON.stringify({ error: error.message }), { status: 500 });
  }
}
