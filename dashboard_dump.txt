"use client";

interface DataPoint {
  date: string;
  total: number;
}

export function DashboardClient({ chart }: { chart: DataPoint[] }) {
  if (chart.length === 0) {
    return <p className="text-sm text-muted-foreground">No sales data yet</p>;
  }

  const max = Math.max(...chart.map((d) => Number(d.total)), 1);

  return (
    <div className="h-56 flex items-end gap-1">
      {chart.map((d) => {
        const height = max > 0 ? (Number(d.total) / max) * 100 : 0;
        return (
          <div key={d.date}
            className="flex-1 flex flex-col items-center group relative"
            title={`${d.date}: $${Number(d.total).toFixed(2)}`}>
            <div className="w-full bg-primary rounded-t transition-opacity group-hover:opacity-80"
              style={{ height: `${Math.max(height, 2)}%` }} />
            <div className="absolute -top-8 hidden group-hover:block bg-popover border rounded px-2 py-1 text-xs whitespace-nowrap shadow">
              {d.date}: ${Number(d.total).toFixed(2)}
            </div>
          </div>
        );
      })}
    </div>
  );
}

import { prisma } from "@/lib/db";
import { success, errorResponse } from "@/lib/api-response";
import { requirePermission } from "@/lib/auth-guard";

export async function GET() {
  try {
    await requirePermission("reports.view");

    const now = new Date();
    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
    const startOfLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);

    const [
      totalRevenue,
      monthRevenue,
      lastMonthRevenue,
      totalOrders,
      monthOrders,
      totalCustomers,
      totalProducts,
      pendingOrders,
      lowStock,
      recentOrders,
      salesByDay,
    ] = await Promise.all([
      prisma.order.aggregate({
        where: { status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] } },
        _sum: { total: true },
      }),
      prisma.order.aggregate({
        where: {
          placedAt: { gte: startOfMonth },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        _sum: { total: true },
      }),
      prisma.order.aggregate({
        where: {
          placedAt: { gte: startOfLastMonth, lt: startOfMonth },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        _sum: { total: true },
      }),
      prisma.order.count(),
      prisma.order.count({ where: { placedAt: { gte: startOfMonth } } }),
      prisma.user.count({ where: { role: "CUSTOMER" } }),
      prisma.product.count({ where: { status: "ACTIVE" } }),
      prisma.order.count({ where: { status: "PENDING" } }),
      prisma.productVariant.count({
        where: { stock: { lte: 5 }, product: { status: "ACTIVE" } },
      }),
      prisma.order.findMany({
        take: 5,
        orderBy: { placedAt: "desc" },
        select: {
          id: true,
          orderNumber: true,
          total: true,
          status: true,
          placedAt: true,
          user: { select: { name: true, email: true } },
        },
      }),
      // Sales by day (last 30 days)
      prisma.order.findMany({
        where: {
          placedAt: { gte: new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000) },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        select: { placedAt: true, total: true },
      }),
    ]);

    // Aggregate sales by day
    const salesMap = new Map<string, number>();
    for (const o of salesByDay) {
      const day = o.placedAt.toISOString().slice(0, 10);
      salesMap.set(day, (salesMap.get(day) ?? 0) + Number(o.total));
    }
    const salesChart = Array.from(salesMap.entries())
      .map(([date, total]) => ({ date, total }))
      .sort((a, b) => a.date.localeCompare(b.date));

    const revenueNow = Number(totalRevenue._sum.total ?? 0);
    const monthNow = Number(monthRevenue._sum.total ?? 0);
    const monthPrev = Number(lastMonthRevenue._sum.total ?? 0);
    const revenueTrend = monthPrev > 0 ? ((monthNow - monthPrev) / monthPrev) * 100 : 0;

    return success({
      revenue: { total: revenueNow, thisMonth: monthNow, trend: revenueTrend },
      orders: { total: totalOrders, thisMonth: monthOrders, pending: pendingOrders },
      customers: totalCustomers,
      products: totalProducts,
      lowStockCount: lowStock,
      recentOrders,
      salesChart,
    });
  } catch (err) {
    return errorResponse(err);
  }
}

    register,
    handleSubmit,
    setValue,
    watch,
    reset,
    formState: { errors },
  } = useForm<FormValues>({
    resolver: zodResolver(schema) as any,

      name: product?.name ?? "",
      description: product?.description ?? "",
      shortDescription: product?.shortDescription ?? "",
      basePrice: product ? Number(product.basePrice) : 0,
      compareAtPrice: product?.compareAtPrice
        ? Number(product.compareAtPrice)

      sku: product?.sku ?? "",
      categoryId: product?.categoryId ?? "",
      status: product?.status ?? "DRAFT",
      isFeatured: product?.isFeatured ?? false,
      stock: product?.stock ?? 0,

      metaTitle: product?.metaTitle ?? "",

    },

  const status = watch("status");
  const isFeatured = watch("isFeatured");

    if (!files) return;
    setUploading(true);
    try {
      const newUrls: string[] = [];
      for (const file of Array.from(files)) {
        const fd = new FormData();
        fd.append("file", file);
        const res = await fetch("/api/v1/admin/upload", {
          method: "POST",
          body: fd,
        });
        if (!res.ok) throw new Error("Upload failed");
        const { data } = await res.json();
        newUrls.push(data.url);
      }
      setImages((prev) => [...prev, ...newUrls]);
      toast.success(`${newUrls.length} image(s) uploaded`);
    } catch {
      toast.error("Upload failed");
    } finally {
      setUploading(false);
    }
  };

  const onSubmit = async (data: FormValues) => {
    setLoading(true);
    try {
      const url = product
        ? `/api/v1/admin/products/${product.id}`
        : "/api/v1/admin/products";
      const method = product ? "PATCH" : "POST";

      const payload = {
        ...data,
        images,
        compareAtPrice: data.compareAtPrice || undefined,
      };

      const res = await fetch(url, {
        method,
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload),
      });

      if (!res.ok) {

      toast.error(err.message ?? "Something went wrong");
    } finally {
      setLoading(false);
    }
