const schema = z.object({

  slug: z.string().optional(),
  description: z.string().optional(),

  parentId: z.string().nullable().optional(),

  isActive: z.boolean().default(true),

// GET /api/v1/admin/categories
export async function GET(request: NextRequest) {
  try {
    await requirePermission("categories.view");

    const page = Math.max(1, Number(request.nextUrl.searchParams.get("page") ?? 1));
    const pageSize = Math.min(
      100,
      Math.max(1, Number(request.nextUrl.searchParams.get("pageSize") ?? 50))
    );
    const q = request.nextUrl.searchParams.get("q") ?? "";

    const where = q
      ? {
          OR: [
            { name: { contains: q, mode: "insensitive" as const } },
            { slug: { contains: q, mode: "insensitive" as const } },
          ],
        }
      : {};

    const [categories, total] = await Promise.all([
      prisma.category.findMany({
        where,
        orderBy: [{ order: "asc" }, { name: "asc" }],
        skip: (page - 1) * pageSize,
        take: pageSize,
        include: {
          parent: { select: { id: true, name: true } },
          _count: { select: { products: true, children: true } },
        },
      }),
      prisma.category.count({ where }),
    ]);

    return paginated(categories, total, page, pageSize);
  } catch (err) {
    return errorResponse(err);
  }
}

// POST /api/v1/admin/categories
export async function POST(request: NextRequest) {
  try {
    const user = await requirePermission("categories.manage");
    const body = schema.parse(await request.json());

    const slug =
      body.slug ??
      body.name
        .toLowerCase()
        .replace(/[^a-z0-9]+/g, "-")
        .replace(/^-|-$/g, "");

    const category = await prisma.category.create({
      data: {
        name: body.name,
        slug,
        description: body.description,
        image: body.image,
        parentId: body.parentId ?? null,
        order: body.order,
        isActive: body.isActive,
      },
    });

    await logAudit({
      userId: user.id,
      action: "CREATE",
      resource: "category",
      resourceId: category.id,
      metadata: { name: category.name },
      ...getRequestInfo(request),
    });

    return success({ category }, 201);
  } catch (err) {
    return errorResponse(err);
  }
}

import { NextRequest } from "next/server";

import { NotFoundError, AppError } from "@/lib/errors";
import { z } from "zod";

type Params = { params: Promise<{ id: string }> };

const updateSchema = z.object({
  name: z.string().min(2).optional(),
  slug: z.string().min(2).optional(),
  description: z.string().nullable().optional(),
  image: z.string().nullable().optional(),
  parentId: z.string().nullable().optional(),
  order: z.number().int().optional(),
  isActive: z.boolean().optional(),
});

// GET /api/v1/admin/categories/:id
export async function GET(_req: NextRequest, { params }: Params) {
  try {
    await requirePermission("categories.view");
    const { id } = await params;

    const category = await prisma.category.findUnique({
      where: { id },
      include: {
        parent: { select: { id: true, name: true } },
        children: { select: { id: true, name: true, slug: true } },
        _count: { select: { products: true, children: true } },
      },
    });

    if (!category) throw new NotFoundError("Category");

    return success({ category });
  } catch (err) {
    return errorResponse(err);
  }
}

// PATCH /api/v1/admin/categories/:id
export async function PATCH(request: NextRequest, { params }: Params) {
  try {
    const user = await requirePermission("categories.manage");
    const { id } = await params;
    const body = updateSchema.parse(await request.json());

    const existing = await prisma.category.findUnique({ where: { id } });
    if (!existing) throw new NotFoundError("Category");

    if (body.parentId === id) {
      throw new AppError("Category cannot be its own parent", 400);
    }

    const updated = await prisma.category.update({
      where: { id },
      data: body,
    });

    await logAudit({
      userId: user.id,
      action: "UPDATE",
      resource: "category",
      resourceId: id,
      metadata: { name: updated.name },
      ...getRequestInfo(request),
    });

    return success({ category: updated });
  } catch (err) {
    return errorResponse(err);
  }
}

    if (existing._count.products > 0) {
      throw new AppError(
        `Cannot delete: ${existing._count.products} products are in this category`,
        400
      );
    }

    if (existing._count.children > 0) {
      throw new AppError(
        `Cannot delete: ${existing._count.children} subcategories exist`,
        400
      );
    }

    await prisma.category.delete({ where: { id } });

    await logAudit({
      userId: user.id,
      action: "DELETE",
      resource: "category",
      resourceId: id,
      metadata: { name: existing.name },
      ...getRequestInfo(request),
    });

    return success({ deleted: true });
  } catch (err) {
    return errorResponse(err);
  }
}

import { NextRequest } from "next/server";
import { success, errorResponse } from "@/lib/api-response";
import { requirePermission } from "@/lib/auth-guard";
import { saveUpload } from "@/lib/upload";
import { logAudit } from "@/lib/audit";
import { z } from "zod";

export async function POST(request: NextRequest) {
  try {
    await requirePermission("products.manage");
    const formData = await request.formData();
    const file = formData.get("file");

    const schema = z.instanceof(File).refine((f) => f.size > 0, "File required")
      .refine((f) => f.size <= 5 * 1024 * 1024, "Max 5MB")
      .refine(
        (f) => ["image/jpeg", "image/png", "image/webp"].includes(f.type),
        "JPEG, PNG, or WebP only"
      );

    const parsed = schema.safeParse(file);
    if (!parsed.success) {
      return Response.json(
        { success: false, error: { code: "VALIDATION_ERROR", message: parsed.error.message } },
        { status: 400 }
      );
    }

    const url = await saveUpload(parsed.data);
    return success({ url });
  } catch (err) {
    return errorResponse(err);
  }
}

import { writeFile, mkdir } from "fs/promises";
import { join } from "path";
import { existsSync } from "fs";

const UPLOAD_DIR = join(process.cwd(), "public", "uploads");

export async function saveUpload(file: File): Promise<string> {
  if (!existsSync(UPLOAD_DIR)) {
    await mkdir(UPLOAD_DIR, { recursive: true });
  }
  const bytes = await file.arrayBuffer();
  const buffer = Buffer.from(bytes);
  const ext = file.name.split(".").pop() ?? "jpg";
  const filename = `${Date.now()}-${Math.random().toString(36).slice(2, 8)}.${ext}`;
  const fullPath = join(UPLOAD_DIR, filename);
  await writeFile(fullPath, buffer);
  return `/uploads/${filename}`;
}
