export async function GET() {
  try {
    await requirePermission("reports.view");

    const now = new Date();
    const startOfMonth = new Date(now.getFullYear(), now.getMonth(), 1);
    const startOfLastMonth = new Date(now.getFullYear(), now.getMonth() - 1, 1);

    const [
      totalRevenue,
      monthRevenue,
      lastMonthRevenue,
      totalOrders,
      monthOrders,
      totalCustomers,
      totalProducts,
      pendingOrders,
      lowStock,
      recentOrders,
      salesByDay,
    ] = await Promise.all([
      prisma.order.aggregate({
        where: { status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] } },
        _sum: { total: true },
      }),
      prisma.order.aggregate({
        where: {
          placedAt: { gte: startOfMonth },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        _sum: { total: true },
      }),
      prisma.order.aggregate({
        where: {
          placedAt: { gte: startOfLastMonth, lt: startOfMonth },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        _sum: { total: true },
      }),
      prisma.order.count(),
      prisma.order.count({ where: { placedAt: { gte: startOfMonth } } }),
      prisma.user.count({ where: { role: "CUSTOMER" } }),
      prisma.product.count({ where: { status: "ACTIVE" } }),
      prisma.order.count({ where: { status: "PENDING" } }),
      prisma.productVariant.count({
        where: { stock: { lte: 5 }, product: { status: "ACTIVE" } },
      }),
      prisma.order.findMany({
        take: 5,
        orderBy: { placedAt: "desc" },
        select: {
          id: true,
          orderNumber: true,
          total: true,
          status: true,
          placedAt: true,
          user: { select: { name: true, email: true } },
        },
      }),
      // Sales by day (last 30 days)
      prisma.order.findMany({
        where: {
          placedAt: { gte: new Date(now.getTime() - 30 * 24 * 60 * 60 * 1000) },
          status: { in: ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"] },
        },
        select: { placedAt: true, total: true },
      }),
    ]);

    // Aggregate sales by day
    const salesMap = new Map<string, number>();
    for (const o of salesByDay) {
      const day = o.placedAt.toISOString().slice(0, 10);
      salesMap.set(day, (salesMap.get(day) ?? 0) + Number(o.total));
    }
    const salesChart = Array.from(salesMap.entries())
      .map(([date, total]) => ({ date, total }))
      .sort((a, b) => a.date.localeCompare(b.date));

    const revenueNow = Number(totalRevenue._sum.total ?? 0);
    const monthNow = Number(monthRevenue._sum.total ?? 0);
    const monthPrev = Number(lastMonthRevenue._sum.total ?? 0);
    const revenueTrend = monthPrev > 0 ? ((monthNow - monthPrev) / monthPrev) * 100 : 0;

    return success({
      revenue: { total: revenueNow, thisMonth: monthNow, trend: revenueTrend },
      orders: { total: totalOrders, thisMonth: monthOrders, pending: pendingOrders },
      customers: totalCustomers,
      products: totalProducts,

      recentOrders,

import { prisma } from "@/lib/db";
import { success, errorResponse } from "@/lib/api-response";
import { requirePermission } from "@/lib/auth-guard";

export const dynamic = "force-dynamic";

const VALID = ["CONFIRMED", "PROCESSING", "SHIPPED", "DELIVERED"];
const DAY = 24 * 60 * 60 * 1000;
const TZ_OFFSET = 6 * 60 * 60 * 1000; // Bangladesh UTC+6

const dayKey = (d: Date) => new Date(d.getTime() + TZ_OFFSET).toISOString().slice(0, 10);

async function loadPeriod(from: Date, to: Date) {
  const orders = await prisma.order.findMany({
    where: { placedAt: { gte: from, lt: to }, status: { in: VALID } },
    select: {
      placedAt: true,
      total: true,
      items: {
        select: {
          quantity: true,
          price: true,
          variant: { select: { product: { select: { costPrice: true } } } },
        },
      },
    },
  });

  const byDay = new Map<string, { revenue: number; profit: number; units: number }>();
  const totals = { revenue: 0, orders: orders.length, units: 0, profit: 0, missingCost: 0 };

  for (const o of orders) {
    const key = dayKey(o.placedAt);
    const row = byDay.get(key) ?? { revenue: 0, profit: 0, units: 0 };
    const orderTotal = Number(o.total);
    row.revenue += orderTotal;
    totals.revenue += orderTotal;

    for (const it of o.items) {
      row.units += it.quantity;
      totals.units += it.quantity;
      const cost = it.variant?.product?.costPrice;
      if (cost === null || cost === undefined) {
        totals.missingCost += 1;
      } else {
        const p = (Number(it.price) - Number(cost)) * it.quantity;
        row.profit += p;
        totals.profit += p;
      }
    }
    byDay.set(key, row);
  }
  return { byDay, totals };
}

export async function GET() {
  try {
    await requirePermission("reports.view");

    const now = new Date();
    const start = new Date(now.getTime() - 30 * DAY);
    const prevStart = new Date(now.getTime() - 60 * DAY);

    const [cur, prev] = await Promise.all([
      loadPeriod(start, new Date(now.getTime() + 1000)),
      loadPeriod(prevStart, start),
    ]);

    const days = [];
    for (let i = 29; i >= 0; i--) {
      const key = dayKey(new Date(now.getTime() - i * DAY));
      const r = cur.byDay.get(key) ?? { revenue: 0, profit: 0, units: 0 };
      days.push({ date: key, revenue: r.revenue, profit: r.profit, units: r.units });
    }

    return success({
      days,
      current: cur.totals,
      previous: prev.totals,
      updatedAt: now.toISOString(),
    });
  } catch (err) {
    return errorResponse(err);
  }
}

import { NextRequest } from "next/server";
import { prisma } from "@/lib/db";
import { success, paginated, errorResponse } from "@/lib/api-response";
import { requirePermission } from "@/lib/auth-guard";
import { Prisma } from "@prisma/client";
import { z } from "zod";

const filterSchema = z.object({
  status: z.string().optional(),
  q: z.string().optional(),
});

export async function GET(request: NextRequest) {
  try {
    await requirePermission("orders.view");
    const raw = Object.fromEntries(request.nextUrl.searchParams);
    const page = Math.max(1, Number(request.nextUrl.searchParams.get("page") ?? 1));
    const pageSize = Math.min(100, Math.max(1, Number(request.nextUrl.searchParams.get("pageSize") ?? 20)));
    const filters = filterSchema.parse(raw);

    const where: Prisma.OrderWhereInput = {
      ...(filters.status && { status: filters.status }),
      ...(filters.q && {
        OR: [
          { orderNumber: { contains: filters.q, mode: "insensitive" } },
          { user: { email: { contains: filters.q, mode: "insensitive" } } },
          { user: { name: { contains: filters.q, mode: "insensitive" } } },
        ],
      }),
    };

    const [orders, total] = await Promise.all([
      prisma.order.findMany({
        where,
        orderBy: { placedAt: "desc" },
        skip: (page - 1) * pageSize,
        take: pageSize,
        include: {
          user: { select: { name: true, email: true } },
          _count: { select: { items: true } },
        },
      }),
      prisma.order.count({ where }),
    ]);

    return paginated(orders, total, page, pageSize);
  } catch (err) {
    return errorResponse(err);
  }
}

              id: true,
              name: true,
              slug: true,
              status: true,
              images: { take: 1, where: { isPrimary: true } },

const patchSchema = z.object({
  id: z.string().min(1),
  status: z.enum(["REQUESTED", "APPROVED", "REJECTED", "COMPLETED"]),
});

export async function PATCH(request: NextRequest) {
  try {
    const user = await requirePermission("returns.manage");
    const parsed = patchSchema.safeParse(await request.json());

    if (!parsed.success) {
      throw new AppError("Invalid return status request.", 400, "INVALID_REQUEST");
    }

    const { id, status } = parsed.data;

    const returnRequest = await prisma.return.findUnique({
      where: { id },
      include: {
        order: {
          include: {
            user: true,
            payments: true,
          },
        },
        refunds: true,
      },
    });

    if (!returnRequest) throw new NotFoundError("Return");

    if (returnRequest.status === "COMPLETED") {
      if (status === "COMPLETED") {
        return success({ return: returnRequest, message: "Refund already completed." });
      }
      throw new AppError(
        "A completed return cannot be changed.",
        409,
        "RETURN_ALREADY_COMPLETED"
      );
    }

    if (status === "COMPLETED") {
      throw new AppError(
        "A return can only be completed after Stripe confirms the refund.",
        400,
        "REFUND_CONFIRMATION_REQUIRED"
      );
    }

    if (status === "REQUESTED") {
      if (returnRequest.status !== "REQUESTED") {
        throw new AppError(
          "This return cannot be moved back to Requested.",
          409,
          "INVALID_RETURN_TRANSITION"
        );
      }
      return success({ return: returnRequest });
    }

    if (status === "REJECTED") {
      if (returnRequest.status !== "REQUESTED") {
        throw new AppError(
          "Only a requested return can be rejected.",
          409,
          "INVALID_RETURN_TRANSITION"
        );
      }

      const updated = await prisma.return.update({
        where: { id },
        data: { status: "REJECTED" },
      });

      await logAudit({
        userId: user.id,
        action: "UPDATE_STATUS",
        resource: "return",
        resourceId: id,
        metadata: { status: "REJECTED" },
        ...getRequestInfo(request),
      });

      return success({ return: updated, message: "Return request rejected." });
    }

    // APPROVED: initiate or safely resume the Stripe refund.
    if (returnRequest.status !== "REQUESTED" && returnRequest.status !== "APPROVED") {
      throw new AppError(
        "Only a requested or already approved return can be processed.",
        409,
        "INVALID_RETURN_TRANSITION"
      );
    }

    if (!isStripeConfigured() || !stripe) {
      throw new AppError(
        "Stripe is not configured. No refund was issued.",
        503,
        "PAYMENT_NOT_CONFIGURED"
      );
    }

    const payment = returnRequest.order.payments.find(
      (item) =>
        item.provider === "stripe" &&
        (item.status === "SUCCEEDED" || item.status === "PARTIALLY_REFUNDED") &&
        item.providerPaymentId.startsWith("pi_")
    );

    if (!payment) {
      throw new AppError(
        "No successful Stripe payment was found for this order. No refund was issued.",
        400,
        "NO_REFUNDABLE_STRIPE_PAYMENT"
      );
    }

    const amount = Number(returnRequest.refundAmount);
    if (!Number.isFinite(amount) || amount <= 0) {
      throw new AppError("The refund amount is invalid.", 400, "INVALID_REFUND_AMOUNT");
    }

    // Reuse an existing refund for this return rather than issuing another one.
    const existingRefund = returnRequest.refunds.find(
      (item) => item.paymentId === payment.id
    );

    if (existingRefund?.status === "SUCCEEDED") {
      const updated = await prisma.return.update({
        where: { id },
        data: { status: "COMPLETED" },
      });

      await logAudit({
        userId: user.id,
        action: "UPDATE_STATUS",
        resource: "return",
        resourceId: id,
        metadata: { status: "COMPLETED", refundId: existingRefund.id },
        ...getRequestInfo(request),
      });

      return success({ return: updated, message: "Refund was already successful." });
    }

    if (existingRefund?.providerRefundId && existingRefund.status === "PENDING") {
      const providerRefund = await stripe.refunds.retrieve(
        existingRefund.providerRefundId
      );

      if (providerRefund.status === "succeeded") {
        await prisma.refund.update({
          where: { id: existingRefund.id },
          data: { status: "SUCCEEDED" },
        });

        const updated = await prisma.return.update({
          where: { id },
          data: { status: "COMPLETED" },
        });

        return success({
          return: updated,
          message: "Stripe confirmed that the refund succeeded.",
        });
      }

      if (
        providerRefund.status === "pending" ||
        providerRefund.status === "requires_action"
      ) {
        const updated = await prisma.return.update({
          where: { id },
          data: { status: "APPROVED" },
        });

        return success({
          return: updated,
          message: "The Stripe refund is still pending.",
        });
      }

      await prisma.refund.update({
        where: { id: existingRefund.id },
        data: { status: "FAILED" },
      });

      throw new AppError(
        "The previous Stripe refund did not succeed. Review the refund before retrying.",
        409,
        "PREVIOUS_REFUND_FAILED"
      );
    }

    // Include completed and pending refunds so the payment cannot be over-refunded.
    const paymentRefunds = await prisma.refund.findMany({
      where: {
        paymentId: payment.id,
        status: { in: ["SUCCEEDED", "PENDING"] },
      },
      select: { amount: true },
    });

    const alreadyReserved = paymentRefunds.reduce(
      (total, refund) => total + Number(refund.amount),
      0
    );

    if (alreadyReserved + amount > Number(payment.amount) + 0.001) {
      throw new AppError(
        "This refund would exceed the remaining refundable payment amount.",
        400,
        "REFUND_TOO_LARGE"
      );
    }

    // Persist approval before calling Stripe.
    if (returnRequest.status !== "APPROVED") {
      await prisma.return.update({
        where: { id },
        data: { status: "APPROVED" },
      });
    }

    const refund = await stripe.refunds.create(
      {
        payment_intent: payment.providerPaymentId,
        amount: Math.round(amount * 100),
        reason: "requested_by_customer",
        metadata: {
          returnId: returnRequest.id,
          orderId: returnRequest.orderId,
          paymentId: payment.id,
        },
      },
      {
        idempotencyKey: `return-refund-${returnRequest.id}`,
      }
    );

    const refundStatus =
      refund.status === "succeeded"
        ? "SUCCEEDED"
        : refund.status === "failed" || refund.status === "canceled"
          ? "FAILED"
          : "PENDING";

    const savedRefund = await prisma.refund.create({
      data: {
        paymentId: payment.id,
        returnId: returnRequest.id,
        amount: new Decimal(amount.toFixed(2)),
        status: refundStatus,
        providerRefundId: refund.id,
        reason: returnRequest.reason,
      },
    });

    let updatedReturn;

    if (refundStatus === "SUCCEEDED") {
      const successfulRefunds = await prisma.refund.findMany({
        where: { paymentId: payment.id, status: "SUCCEEDED" },
        select: { amount: true },
      });

      const refundedTotal = successfulRefunds.reduce(
        (total, item) => total + Number(item.amount),
        0
      );

      const fullyRefunded = refundedTotal >= Number(payment.amount) - 0.001;

      await prisma.payment.update({
        where: { id: payment.id },
        data: {
          status: fullyRefunded ? "REFUNDED" : "PARTIALLY_REFUNDED",
          refundedAt: fullyRefunded ? new Date() : null,
        },
      });

      if (fullyRefunded) {
        await prisma.order.update({
          where: { id: returnRequest.orderId },
          data: { status: "REFUNDED" },
        });
      }

      updatedReturn = await prisma.return.update({
        where: { id },
        data: { status: "COMPLETED" },
      });
    } else {
      updatedReturn = await prisma.return.update({
        where: { id },
        data: { status: "APPROVED" },
      });
    }

    await logAudit({
      userId: user.id,
      action: "UPDATE_STATUS",
      resource: "return",
      resourceId: id,
      metadata: {
        status: updatedReturn.status,
        refundId: savedRefund.id,
        providerRefundId: refund.id,
        stripeStatus: refund.status,
        amount,
      },
      ...getRequestInfo(request),
    });

    return success({
      return: updatedReturn,
      refund: {
        id: savedRefund.id,
        status: refundStatus,
        providerStatus: refund.status,
        amount,
      },
      message:
        refundStatus === "SUCCEEDED"
          ? "Stripe refund completed successfully."
          : refundStatus === "PENDING"
            ? "Refund submitted to Stripe and is pending confirmation."
            : "Stripe did not complete the refund. Review the refund record.",
    });
  } catch (err) {
    return errorResponse(err);
  }
}

        product.shortDescription ?? "",
        product.description ?? "",
      ].join(" ").toLocaleLowerCase();
      const category = (product.category?.name ?? "").toLocaleLowerCase();
      const combined = `${name} ${category} ${description}`;

      const variantPrices = product.variants.map((variant) =>
        Number(variant.price)
      );

      const price =
        variantPrices.length > 0
          ? Math.min(...variantPrices)
          : Number(product.basePrice);

      const variantStock = product.variants.reduce(
        (total, variant) => total + Math.max(0, variant.stock),
        0
      );

      const hasActiveVariants = product.variants.length > 0;
      const stock = hasActiveVariants ? variantStock : product.stock;
      const inStock = product.trackInventory ? stock > 0 : true;

      const score = terms.reduce((total, term) => {
        if (name === term) return total + 12;
        if (name.includes(term)) return total + 7;
        if (category.includes(term)) return total + 4;
        if (description.includes(term)) return total + 1;
        return total;
      }, 0);

      return {
        product,
        score,
        price,
        stock,
        inStock,
        variantPrices,
        combined,
        name,
        category,
      };
    })
    .filter((item) => {
      if (item.score <= 0) return false;


      // Gift recommendations must be plausible gift items.
      // Do not turn medicated products into casual gift suggestions.
      if (intent === "gift") {
        if (excludedGiftTerms.test(item.combined)) return false;
        if (!giftTerms.test(`${item.name} ${item.category}`)) return false;
      }

      return true;
    })
    .sort(
      (a, b) =>
        Number(b.inStock) - Number(a.inStock) ||
        b.score - a.score ||
        a.price - b.price
    )
    .slice(0, 8);

  // Avoid presenting irrelevant results when a suitable gift is not found.
  return scored.map(({ product, price, stock, inStock }) => ({
    name: product.name,
    price,
    comparePrice: product.compareAtPrice
      ? Number(product.compareAtPrice)
      : null,
    category: product.category?.name ?? null,
    inStock,
    stock: product.trackInventory ? stock : null,
    description: String(
      product.shortDescription ?? product.description ?? ""
    ).slice(0, 400),
    url: "/product/" + product.slug,
    image: product.images[0]?.url ?? null,
  }));
}

async function listCategories() {
  const all = await prisma.category.findMany({
    where: { isActive: true },
    orderBy: [{ order: "asc" }, { name: "asc" }],
    select: { id: true, name: true, slug: true, parentId: true },
  });
  return all
    .filter((c) => !c.parentId)
    .map((c) => ({
      name: c.name,
      url: "/shop?category=" + c.slug,
      subCategories: all
        .filter((s) => s.parentId === c.id)
        .map((s) => s.name)
        .slice(0, 15),
    }));
}
