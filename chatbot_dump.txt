import { groq } from "@ai-sdk/groq";
import { streamText, tool, stepCountIs, convertToModelMessages } from "ai";
import { z } from "zod";
import { Prisma } from "@prisma/client";
import { prisma } from "@/lib/db";
import { setDefaultResultOrder } from "node:dns";

setDefaultResultOrder("ipv4first");

export const dynamic = "force-dynamic";
export const maxDuration = 30;

async function searchProducts(query: string) {
  const terms = query
    .split(/\s+/)
    .map((t) => t.trim())
    .filter((t) => t.length >= 2)
    .slice(0, 5);

  const or: Prisma.ProductWhereInput[] = terms.flatMap((t) => [
    { name: { contains: t, mode: "insensitive" } },
    { description: { contains: t, mode: "insensitive" } },
    { category: { name: { contains: t, mode: "insensitive" } } },
  ]);

  const where: Prisma.ProductWhereInput = { status: "ACTIVE", ...(or.length > 0 ? { OR: or } : {}) };

  const rows = await prisma.product.findMany({
    where,
    orderBy: { createdAt: "desc" },
    take: 6,
    include: {
      category: { select: { name: true } },
      images: { where: { isPrimary: true }, take: 1 },
    },
  });

  const ids = rows.map((r) => r.id);
  const variants =
    ids.length > 0
      ? await prisma.productVariant.findMany({
          where: { productId: { in: ids }, isActive: true },
          select: { productId: true, stock: true },
        })
      : [];
  const stock = new Map<string, number>();
  for (const v of variants) stock.set(v.productId, (stock.get(v.productId) ?? 0) + (v.stock ?? 0));

  return rows.map((p) => {
    const has = stock.has(p.id);
    const total = stock.get(p.id) ?? 0;
    return {
      name: p.name,
      price: Number(p.basePrice),
      comparePrice: p.compareAtPrice ? Number(p.compareAtPrice) : null,
      category: p.category?.name ?? null,
      inStock: has ? total > 0 : null,
      stock: has ? total : null,
      description: String(p.description ?? "").slice(0, 300),
      url: "/product/" + p.slug,
      image: p.images?.[0]?.url ?? null,
    };
  });
}

async function listCategories() {
  const all = await prisma.category.findMany({
    where: { isActive: true },
    orderBy: [{ order: "asc" }, { name: "asc" }],
    select: { id: true, name: true, slug: true, parentId: true },
  });
  return all
    .filter((c) => !c.parentId)
    .map((c) => ({
      name: c.name,
      url: "/shop?category=" + c.slug,
      subCategories: all
        .filter((s) => s.parentId === c.id)
        .map((s) => s.name)
        .slice(0, 15),
    }));
}

export async function POST(req: Request) {
  try {
    const { messages } = await req.json();

    const result = streamText({
      model: groq("openai/gpt-oss-120b"),
      system:
        "You are T-bot, a polite and helpful customer support assistant for Trillion Mart (an online store in Bangladesh). " +
        "For ANY question about products, prices, discounts, availability or stock, or what the store sells, call the searchProducts tool. " +
        "For questions about categories, call the listCategories tool. " +
        "Never invent products, prices or stock: answer only from tool results. Prices are in Taka (BDT). " +
        "If nothing is found, say so politely and suggest browsing the shop page (/shop). " +
        "Keep answers short. Always reply in the language the user writes in (Bangla or English).",
      messages: await convertToModelMessages(messages),
      stopWhen: stepCountIs(4),
      tools: {
        searchProducts: tool({
          description:
            "Search the store database for active products by name, keyword or category. Returns price, discount, stock and link.",
          inputSchema: z.object({
            query: z.string().describe("Product name or keyword, e.g. 'speaker' or 'vitamin c serum'"),
          }),
          execute: async ({ query }) => {
            const products = await searchProducts(query);
            return { products };
          },
        }),
        listCategories: tool({
          description: "List the store's main categories with their sub-categories.",
          inputSchema: z.object({}),
          execute: async () => {
            const categories = await listCategories();
            return { categories };
          },
        }),
      },
    });

    return result.toUIMessageStreamResponse({
      onError: (e) => (e instanceof Error ? e.message : String(e)),
    });
  } catch (error: any) {
    console.error("API Error:", error);
    return new Response(JSON.stringify({ error: error.message }), { status: 500 });
  }
}

import { useEffect, useRef, useState } from "react";
import Link from "next/link";
import { useSession } from "next-auth/react";
import { ChevronDown } from "lucide-react";
import { SignOutConfirm } from "@/components/auth/sign-out-confirm";

export function HeaderAccount() {
  const { data: session, status } = useSession();
  const [open, setOpen] = useState(false);
  const [confirm, setConfirm] = useState(false);
  const ref = useRef<HTMLDivElement>(null);

  useEffect(() => {
    const onDown = (e: MouseEvent) => {
      if (ref.current && !ref.current.contains(e.target as Node)) setOpen(false);
    };
    document.addEventListener("mousedown", onDown);
    return () => document.removeEventListener("mousedown", onDown);
  }, []);

  const openAuth = (mode: "login" | "register") =>
    window.dispatchEvent(new CustomEvent("open-auth-modal", { detail: mode }));

  if (status === "loading") return <span className="h-4 w-24" />;

  if (status !== "authenticated") {
    return (
      <>
        <button type="button" onClick={() => openAuth("login")} className="hover:underline">LOGIN</button>
        <button type="button" onClick={() => openAuth("register")} className="hover:underline">SIGN UP</button>
      </>
    );
  }

  const first = (session?.user?.name || session?.user?.email || "My").split(/[\s@]/)[0];
  const role = (session?.user as any)?.role as string | undefined;

  return (
    <div ref={ref} className="relative">
      <button type="button" onClick={() => setOpen((o) => !o)} className="flex items-center gap-1 uppercase hover:underline">
        {first}&apos;S ACCOUNT <ChevronDown className="h-3.5 w-3.5" />
      </button>
      {open && (
        <div className="absolute right-0 top-full z-[140] mt-2 w-52 overflow-hidden rounded-md border bg-white p-3 text-gray-800 shadow-2xl normal-case">
          <p className="mb-2 truncate text-sm font-semibold">{session?.user?.name || session?.user?.email}</p>
          <Link
            href="/account"
            onClick={() => setOpen(false)}
            className="block rounded bg-[#f85606] py-2 text-center text-sm font-bold text-white hover:bg-[#d94b05]"
          >
            My Account
          </Link>
          <Link href="/account/orders" onClick={() => setOpen(false)} className="mt-1 block rounded px-2 py-2 text-sm hover:bg-orange-50">
            My Orders
          </Link>
          {(role === "ADMIN" || role === "STAFF") && (
            <Link href="/admin" onClick={() => setOpen(false)} className="block rounded px-2 py-2 text-sm hover:bg-orange-50">
              Admin Panel
            </Link>
          )}
