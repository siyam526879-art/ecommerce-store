  const product = await prisma.product.findUnique({
    where: { slug, status: "ACTIVE" },
    select: { name: true, metaTitle: true, metaDescription: true },

import { useState } from "react";
import { ShoppingCart, Loader2, Check } from "lucide-react";
import { Button } from "@/components/ui/button";
import { QuantitySelector } from "./quantity-selector";
import { useCartStore } from "@/lib/store/cart-store";
import { toast } from "sonner";

interface Variant {
  id: string;
  price: number;
  stock: number;
  sku?: string | null;
  attributes?: Record<string, string>;
}

interface AddToCartButtonProps {
  productId: string;
  productSlug: string;
  productName: string;
  variants: Variant[];
  disabled?: boolean;
  image?: string;
}

export function AddToCartButton({
  productId,
  productSlug,
  productName,
  variants,
  disabled,
  image,
}: AddToCartButtonProps) {
  const [selectedVariantId, setSelectedVariantId] = useState<string>(variants[0]?.id ?? "");
  const [quantity, setQuantity] = useState(1);
  const [adding, setAdding] = useState(false);
  const addItem = useCartStore((s) => s.addItem);

  const selectedVariant = variants.find((v) => v.id === selectedVariantId);
  const currentStock = selectedVariant?.stock ?? 0;

  const handleAdd = () => {
    if (!selectedVariant) return;
    setAdding(true);

    addItem(
      {
        variantId: selectedVariant.id,
        productId,
        name: productName,
        slug: productSlug,
        price: selectedVariant.price,
        image,
        attributes: selectedVariant.attributes,
        stock: selectedVariant.stock,
      },
      quantity
    );

    toast.success("Added to cart");
    setTimeout(() => {
      setAdding(false);
      setQuantity(1);
    }, 500);
  };

  return (
    <div className="space-y-4">
      {variants.length > 1 && (
        <div>
          <label className="text-sm font-medium mb-2 block">Options</label>
          <div className="flex flex-wrap gap-2">
            {variants.map((v) => {
              const label = v.attributes
                ? Object.values(v.attributes).join(" / ")
                : v.sku ?? v.id.slice(0, 8);
              return (
                <button
                  key={v.id}
                  onClick={() => {
                    setSelectedVariantId(v.id);
                    setQuantity(1);
                  }}
                  disabled={v.stock <= 0}
                  className={`px-4 py-2 rounded-md border text-sm transition-colors ${
                    selectedVariantId === v.id
                      ? "border-primary bg-primary text-primary-foreground"
                      : v.stock <= 0
                      ? "border-muted bg-muted text-muted-foreground cursor-not-allowed line-through"
                      : "border-border hover:border-primary"
                  }`}
                >
                  {label}
                </button>
              );
            })}
          </div>
        </div>
      )}

      <div className="flex items-center gap-4">
        <QuantitySelector
          value={quantity}
          onChange={setQuantity}
          max={currentStock}
          disabled={disabled}
        />
        <Button
          onClick={handleAdd}
          disabled={disabled || adding || currentStock <= 0}
          className="flex-1"
          size="lg"
        >
          {adding ? (
            <>
              <Check className="mr-2 h-4 w-4" /> Added
            </>
          ) : (
            <>
              <ShoppingCart className="mr-2 h-4 w-4" />
              {adding ? "Adding..." : "Add to cart"}
            </>
          )}
        </Button>
      </div>
    </div>
  );
}

import { Minus, Plus } from "lucide-react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";

interface QuantitySelectorProps {
  value: number;
  onChange: (value: number) => void;
  min?: number;
  max?: number;
  disabled?: boolean;
}

export function QuantitySelector({
  value,
  onChange,
  min = 1,
  max = 99,
  disabled = false,
}: QuantitySelectorProps) {
  const decrement = () => onChange(Math.max(min, value - 1));
  const increment = () => onChange(Math.min(max, value + 1));

  return (
    <div className="flex items-center border rounded-md w-fit">
      <Button
        type="button"
        variant="ghost"
        size="icon"
        className="h-9 w-9"
        onClick={decrement}
        disabled={disabled || value <= min}
        aria-label="Decrease quantity"
      >
        <Minus className="h-4 w-4" />
      </Button>
      <Input
        type="number"
        value={value}
        onChange={(e) => {
          const v = parseInt(e.target.value) || min;
          onChange(Math.max(min, Math.min(max, v)));
        }}
        className="h-9 w-14 border-0 text-center [appearance:textfield] [&::-webkit-outer-spin-button]:appearance-none [&::-webkit-inner-spin-button]:appearance-none"
        min={min}
        max={max}
        disabled={disabled}
      />
      <Button
        type="button"
        variant="ghost"
        size="icon"
        className="h-9 w-9"
        onClick={increment}
        disabled={disabled || value >= max}
        aria-label="Increase quantity"
      >
        <Plus className="h-4 w-4" />
      </Button>
    </div>
  );
}

"use client";

import { create } from "zustand";
import { persist, createJSONStorage } from "zustand/middleware";

export interface CartItem {
  variantId: string;
  productId: string;
  name: string;
  slug: string;
  price: number;
  image?: string;
  attributes?: Record<string, string>;
  quantity: number;
  stock: number;
}

interface CartState {
  items: CartItem[];
  addItem: (item: Omit<CartItem, "quantity">, quantity?: number) => void;
  removeItem: (variantId: string) => void;
  updateQuantity: (variantId: string, quantity: number) => void;
  clear: () => void;
  itemCount: () => number;
  subtotal: () => number;
}

const CART_VERSION = 6;

// Detect mojibake / corrupted strings
function cleanText(s: string): string {
  if (typeof s !== "string") return "";
  // Remove common mojibake sequences
  const stripped = s.replace(/[][^\s]*/g, "").replace(/[\u0080-\u00BF]+/g, "");
  return stripped.replace(/\s+/g, " ").trim();
}

function isCorruptedString(s: string): boolean {
  if (typeof s !== "string") return true;
  // Reject strings with common mojibake markers
  if (/[]/.test(s)) return true;
  // Reject strings with weird non-ASCII (allow standard letters + emoji ranges)
  if (/[\u0080-\u00FF]{2,}/.test(s)) return true;
  return false;
}

function sanitizeItems(items: any): CartItem[] {
  if (!Array.isArray(items)) return [];
  return items
    .filter((i) => i && typeof i === "object")
    .filter((i) => !isCorruptedString(i.name ?? ""))
    .filter((i) => !isCorruptedString(i.slug ?? ""))
    .map((i) => ({
      variantId: String(i.variantId ?? ""),
      productId: String(i.productId ?? ""),
      name: cleanText(String(i.name ?? "")),
      slug: cleanText(String(i.slug ?? "")),
      price: Number(i.price ?? 0),
      image: typeof i.image === "string" ? i.image : undefined,
      attributes: i.attributes && typeof i.attributes === "object" ? i.attributes : undefined,
      quantity: Math.max(1, Number(i.quantity ?? 1)),
      stock: Math.max(0, Number(i.stock ?? 0)),
    }))
    .filter((i) => i.variantId && i.name);
}

export const useCartStore = create<CartState>()(
  persist(
    (set, get) => ({
      items: [],
      addItem: (item, quantity = 1) => {
        set((state) => {
          const existing = state.items.find((i) => i.variantId === item.variantId);
          if (existing) {
            const newQty = Math.min(existing.quantity + quantity, item.stock);
            return {
              items: state.items.map((i) =>
                i.variantId === item.variantId
                  ? { ...i, quantity: newQty, image: item.image ?? i.image }
                  : i
              ),
            };
          }
          return {
            items: [
              ...state.items,
              { ...item, name: item.name.trim(), quantity: Math.min(quantity, item.stock) },
            ],
          };
        });
      },
      removeItem: (variantId) =>
        set((state) => ({ items: state.items.filter((i) => i.variantId !== variantId) })),
      updateQuantity: (variantId, quantity) =>
        set((state) => ({
          items: state.items.map((i) =>
            i.variantId === variantId
              ? { ...i, quantity: Math.max(1, Math.min(quantity, i.stock)) }
              : i
          ),
        })),
      clear: () => set({ items: [] }),
      itemCount: () => get().items.reduce((sum, i) => sum + i.quantity, 0),
      subtotal: () => get().items.reduce((sum, i) => sum + i.price * i.quantity, 0),
    }),
    {
      name: "ecommerce-cart",
      storage: createJSONStorage(() => localStorage),
      version: CART_VERSION,
      migrate: (persistedState: any, version: number) => {
        if (version < CART_VERSION) {
          // Sanitize instead of full clear
          const clean = sanitizeItems(persistedState?.items);
          return { items: clean };
        }

          email: user.email,
          name: user.name,
          role: user.role,
          image: user.image,
        };
      },
    }),
  ],

    async signIn({ user, account }) {
      if (account?.provider === "google") {
        const email = user.email?.toLowerCase();
        if (!email) return false;

        const existingUser = await prisma.user.findUnique({
          where: { email },
          select: { id: true, isActive: true },
        });

        if (existingUser) {

        const intent = cookieStore.get("auth_intent")?.value;

        });
        if (dbUser) {
          token.role = dbUser.role;
          token.id = dbUser.id;
        }
      }
      return token;
    },

    async session({ session, token }) {
      if (session.user && token) {
        (session.user as { id?: string; role?: string }).id = token.id as string;
        (session.user as { id?: string; role?: string }).role = token.role as string;
      }
      return session;
    },

      // Always redirect to home page after login/register
      return baseUrl;

import { ForbiddenError, UnauthorizedError } from "./errors";
import { Permission, hasPermission } from "./permissions";

export async function requireUser() {
  const session = await auth();
  if (!session?.user?.id) throw new UnauthorizedError("You must be signed in");
  return {
    id: session.user.id as string,
    email: session.user.email as string,
    role: (session.user as any).role as string,
  };
}

export async function requireRole(...roles: string[]) {
  const user = await requireUser();
  if (!roles.includes(user.role)) throw new ForbiddenError("Insufficient permissions");
  return user;
}

export async function requirePermission(permission: Permission) {
  const user = await requireUser();
  if (!hasPermission(user.role, permission)) {
    throw new ForbiddenError(`Missing permission: ${permission}`);
  }
  return user;
}

import { NextRequest } from "next/server";
import { prisma } from "@/lib/db";
import { success, errorResponse } from "@/lib/api-response";
import { requireUser } from "@/lib/auth-guard";
import { z } from "zod";

// Client calls this to validate cart before showing payment UI
const schema = z.object({
  couponCode: z.string().optional(),
});

export async function POST(request: NextRequest) {
  try {
    const user = await requireUser();
    const { couponCode } = schema.parse(await request.json());

    const cart = await prisma.cart.findUnique({
      where: { userId: user.id },
      include: {
        items: {
          include: {
            variant: {
              include: {
                product: { select: { name: true, status: true, trackInventory: true } },
              },
            },
          },
        },
      },
    });

    if (!cart || cart.items.length === 0) {
      return success({ valid: false, error: "Cart is empty" });
    }

    const issues: string[] = [];
    let subtotal = 0;

    for (const item of cart.items) {
      if (item.variant.product.status !== "ACTIVE") {
        issues.push(`"${item.variant.product.name}" is no longer available`);
      } else if (item.variant.product.trackInventory && item.variant.stock < item.quantity) {
        issues.push(`Only ${item.variant.stock} left of "${item.variant.product.name}"`);
      } else {
        subtotal += Number(item.variant.price) * item.quantity;
      }
    }

    // Validate coupon
    let discount = 0;
    if (couponCode) {
      const coupon = await prisma.coupon.findUnique({
        where: { code: couponCode.toUpperCase(), isActive: true },
      });
      if (!coupon) {
        issues.push("Invalid coupon code");
      } else {
        const now = new Date();
        if (coupon.startsAt && coupon.startsAt > now) issues.push("Coupon not yet active");
        if (coupon.expiresAt && coupon.expiresAt < now) issues.push("Coupon has expired");
        if (coupon.maxUses && coupon.usedCount >= coupon.maxUses) issues.push("Coupon usage limit reached");
        if (coupon.minOrder && subtotal < Number(coupon.minOrder)) {
          issues.push(`Minimum order ${Number(coupon.minOrder)} required`);
        }
        if (coupon.type === "PERCENT") discount = (subtotal * Number(coupon.value)) / 100;
        else if (coupon.type === "FIXED") discount = Math.min(Number(coupon.value), subtotal);
      }
    }

    return success({
      valid: issues.length === 0,
      issues,
      subtotal,
      discount,
      itemCount: cart.items.reduce((sum, i) => sum + i.quantity, 0),
    });
  } catch (err) {
    return errorResponse(err);
  }
}
