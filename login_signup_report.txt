export default function LoginPage() {
  const router = useRouter();
  const params = useSearchParams();
  const [loading, setLoading] = useState(false);
  const [successMsg, setSuccessMsg] = useState<string | null>(null);
  const [errorMsg, setErrorMsg] = useState<string | null>(null);

  const {
    register,
    handleSubmit,
    formState: { errors },
  } = useForm<LoginInput>({
    resolver: zodResolver(loginSchema),
  });

  // Handle URL params: google_registered, error, registered
  useEffect(() => {
    const googleReg = params.get("google_registered");
    const err = params.get("error");
    const reg = params.get("registered");

    if (googleReg === "1") {
      // Auto-login after Google registration handled by auth.ts, redirecting...
      router.push("/account");
    } else if (reg === "1") {
      setSuccessMsg("Account created! Please sign in.");
    }

    if (err === "SignupRequired") {
      setErrorMsg("No account found. Please sign up first with email/password or Google on Register page.");
    } else if (err === "AccountDisabled") {
      setErrorMsg("Your account is disabled. Please contact support.");
    } else if (err === "CreateFailed") {
      setErrorMsg("Account creation failed. Please try again.");
    } else if (err === "Configuration") {
      setErrorMsg("Login configuration error. Please contact support.");
    }

    setLoading(true);

      const res = await signIn("credentials", {

      setLoading(false);

function ResetForm() {
  const router = useRouter();

  const [email, setEmail] = useState("");
  const [otp, setOtp] = useState("");
  const [newPassword, setNewPassword] = useState("");
  const [confirm, setConfirm] = useState("");
  const [sessionToken, setSessionToken] = useState("");

  const [step, setStep] = useState<1 | 2>(1);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    const e = params.get("email") ?? "";
    const o = params.get("otp") ?? "";
    setEmail(e);
    if (o) setOtp(o);
  }, [params]);

  // ---- Step 1: Verify OTP ----
  const onVerify = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!email || !otp) {
      toast.error("Please enter OTP");
      return;
    }
    if (otp.length !== 6) {
      toast.error("OTP must be 6 digits");
      return;
    }
    setLoading(true);
    try {
      const res = await fetch("/api/v1/auth/verify-otp", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, otp }),
      });
      const text = await res.text();
      const json = text ? JSON.parse(text) : {};
      if (!res.ok) throw new Error(json.error?.message ?? "Invalid OTP");
      toast.success("OTP verified! Set your new password.");
      setSessionToken(json.data?.sessionToken ?? "");
      setStep(2);
    } catch (err: any) {
      toast.error(err.message ?? "Invalid OTP");
    } finally {
      setLoading(false);
    }
  };

  const onReset = async (e: React.FormEvent) => {
    e.preventDefault();

      toast.error("Please fill both password fields");
      return;
    }
    if (newPassword.length < 8) {
      toast.error("Password must be at least 8 characters");
      return;
    }
    if (newPassword !== confirm) {
      toast.error("Passwords do not match");
      return;
    }
    setLoading(true);
    try {
      const res = await fetch("/api/v1/auth/reset-password", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ email, sessionToken, newPassword }),
      });
      const text = await res.text();
      const json = text ? JSON.parse(text) : {};
      if (!res.ok) throw new Error(json.error?.message ?? "Failed");
      toast.success("Password reset! Redirecting to login...");

export async function requireUser() {
  const session = await auth();
  if (!session?.user?.id) throw new UnauthorizedError("You must be signed in");
  return {
    id: session.user.id as string,
    email: session.user.email as string,
    role: (session.user as any).role as string,
  };
}

export async function requireRole(...roles: string[]) {
  const user = await requireUser();
  if (!roles.includes(user.role)) throw new ForbiddenError("Insufficient permissions");
  return user;
}

export async function requirePermission(permission: Permission) {
  const user = await requireUser();
  if (!hasPermission(user.role, permission)) {
    throw new ForbiddenError(`Missing permission: ${permission}`);
  }
  return user;
}

import { prisma } from "@/lib/db";
import { success, errorResponse } from "@/lib/api-response";
import { z } from "zod";
import crypto from "crypto";

const schema = z.object({
  email: z.string().email(),
  otp: z.string().length(6),
});

const MAX_ATTEMPTS = 5;
const SESSION_TTL_MS = 15 * 60 * 1000;

function hashOTP(otp: string): string {
  return crypto.createHash("sha256").update(otp).digest("hex");
}

function randomSessionToken(): string {
  return crypto.randomBytes(32).toString("hex");
}

export async function POST(req: NextRequest) {
  try {
    const body = schema.parse(await req.json());

    const user = await prisma.user.findUnique({
      where: { email: body.email },
      select: {
        id: true,
        passwordResetToken: true,
        passwordResetExpires: true,
        passwordResetAttempts: true,
      },
    });

    if (!user || !user.passwordResetToken || !user.passwordResetExpires) {
      return errorResponse({
        message: "Invalid or expired OTP. Please request a new one.",
        status: 400,
      });
    }

    if (user.passwordResetAttempts >= MAX_ATTEMPTS) {
      await prisma.user.update({
        where: { id: user.id },
        data: {
          passwordResetToken: null,
          passwordResetExpires: null,
          passwordResetAttempts: 0,
        },
      });
      return errorResponse({
        message: "Too many failed attempts. Please request a new OTP.",
        status: 429,
      });
    }

    if (new Date() > user.passwordResetExpires) {
      return errorResponse({
        message: "OTP has expired. Please request a new one.",
        status: 400,
      });
    }

    const hashed = hashOTP(body.otp);
    if (hashed !== user.passwordResetToken) {
      await prisma.user.update({
        where: { id: user.id },
        data: { passwordResetAttempts: { increment: 1 } },
      });
      const remaining = MAX_ATTEMPTS - (user.passwordResetAttempts + 1);
      return errorResponse({
        message:
          remaining > 0
            ? "Incorrect OTP. " + remaining + " attempts remaining."
            : "Incorrect OTP. No attempts left â€” request a new OTP.",
        status: 400,
      });
    }

    // âœ… OTP correct â€” kill it and give session token
    const sessionToken = randomSessionToken();
    const sessionExpires = new Date(Date.now() + SESSION_TTL_MS);

    await prisma.user.update({
      where: { id: user.id },
      data: {
        passwordResetToken: null,
        passwordResetExpires: null,
        passwordResetAttempts: 0,
        resetSessionToken: sessionToken,
        resetSessionExpires: sessionExpires,
      },
    });

    return success({
      verified: true,
      sessionToken,
      sessionExpires,
    });
  } catch (err) {
